    class BaseModel {
        +String id
    }

    class Appointment {
        +String patientId
        +String dentistId
        +String status
        +Date startAppointmentTime
        +Date endAppointmentTime
        +confirm()
        +start()
        +complete()
        +cancel()
    }

    class Category {
        +String name
    }

    class ClinicalProgres {
        +String patientId
        +String dateId
        +String dentistId
        +String diagnosis
        +Date registrationDate
    }

    class DentalChart {
        +String clinicalProgressId
        +String patientId
        +String dentistId
        +Date evaluationDate
        +String observations
    }

    class DentalChartDetail {
        +String dentalChartId
        +int toothNumber
        +String face
        +String toothStatus
        +String notes
    }

    class Disease {
        +String name
        +String description
    }

    class Installment {
        +String paymentPlanId
        +int installmentNumber
        +Date dueDate
        -number amount
        -number lateFeeAmount
        -number paidAmount
        -status status
        +addPayment(amount)
        +addLateFee(amount)
    }

    class Inventory {
        +String productId
        +String productName
        -int currentStock
        +int minimumStock
        +increase(quantity)
        +decrease(quantity)
    }

    class InventoryLote {
        +String productId
        +String supplierId
        +String loteNumber

        +int quantity
    }

    class InventoryMovement {
        +String productId
        +String movementType
        +int quantity
        +Date movementDate
    }

    class Invoice {
        +String patientId
        +String appointmentId
        +number subtotal
        +number total
        +String status
    }

    class MeasurementUnit {

    }

    class MedicalPrescription {
        +String patientId
        +String dentistId
        +Date prescriptionDate
    }

    class MedicalPrescriptionDetail {
        +String medicalPrescriptionId

        +String patientCode
        +String name
        +String lastName
        +String idCard
        +Date birthdate

        +String email

        +deactivate()

    class PatientDisease {
        +String patientId

    class Product {
        +String name
        +String categoryId
        +String measurementUnitId
        +String supplierId
        -number purchasePrice
        -number salePrice
        -bool active
    }

    class Role {
        +String name
        +String description
    }

    class Supplier {
        +String name
        +String contact
        +String phone
        +String email
    }

        +String code

        -number basePrice

    class TreatmentPlan {
        +String patientId
        +String dentistId
        +String code
        -String status
        -number totalAmount
        +propose()
        +accept()
        +complete()

        +String roleId
        +String fullName
        +String email
        -bool active
