    %% --- FLUJO PRINCIPAL ---
    A --> D_Opcion

    %% ==========================================
    %% 1. GENERAR FACTURA
    %% ==========================================
    D_Opcion -->|Generar factura| U_ReqGenFact
    U_ReqGenFact --> S_ValPerm1
    S_ValPerm1 -->|No| S_NoAuth
    S_ValPerm1 -->|Sí| U_IngPacPlan
    U_IngPacPlan --> S_ConsultPacPlan
    S_ConsultPacPlan --> S_ValFactData
    S_ValFactData --> D_ValData1
    D_ValData1 -->|No| S_ErrVal1
    S_ErrVal1 --> Fin
    D_ValData1 -->|Sí| S_RegFact
    S_RegFact --> S_ShowFact
    S_ShowFact --> Fin

    %% ==========================================
    %% 2. REGISTRAR PAGO DIRECTO
    %% ==========================================
    D_Opcion -->|Registrar pago directo| U_SelFactPago
    U_SelFactPago --> S_ValPerm2
    S_ValPerm2 -->|No| S_NoAuth
    S_ValPerm2 -->|Sí| S_ConsultFact1
    S_ConsultFact1 --> S_ValPago1
    S_ValPago1 --> D_EsTrans1
    
    D_EsTrans1 -->|Sí| U_IngRefTrans1
    U_IngRefTrans1 --> S_ValRef1
    S_ValRef1 --> D_PagoValid1
    
    D_EsTrans1 -->|No| D_PagoValid1
    
    D_PagoValid1 -->|No| S_ErrPago1
    S_ErrPago1 --> Fin
    D_PagoValid1 -->|Sí| S_RegPago1
    S_RegPago1 --> S_ShowPago1
    S_ShowPago1 --> Fin

    %% ==========================================
    %% 3. CREAR PLAN DE PAGOS
    %% ==========================================
    D_Opcion -->|Crear plan de pagos| U_ReqPlan
    U_ReqPlan --> S_ValPerm3
    S_ValPerm3 -->|No| S_NoAuth
    S_ValPerm3 -->|Sí| S_ConsultFact2
    S_ConsultFact2 --> U_DefCuotas
    U_DefCuotas --> S_ValPlanCond
    S_ValPlanCond --> D_PlanValid
    D_PlanValid -->|No| S_ErrPlan
    S_ErrPlan --> Fin
    D_PlanValid -->|Sí| S_CalcCuotas
    S_CalcCuotas --> S_RegPlan
    S_RegPlan --> S_ShowPlan
    S_ShowPlan --> Fin

    %% ==========================================
    %% 4. REGISTRAR PAGO DE CUOTA
    %% ==========================================
    D_Opcion -->|Registrar pago de cuota| U_SelCuota
    U_SelCuota --> S_ValPerm4
    S_ValPerm4 -->|No| S_NoAuth
    S_ValPerm4 -->|Sí| S_ConsultCuota
    S_ConsultCuota --> S_ValCuotaCond
    S_ValCuotaCond --> D_CuotaValid
    
    D_CuotaValid -->|No| S_ErrCuota
    S_ErrCuota --> Fin
    
    D_CuotaValid -->|Sí| U_IngImpMetodo
    U_IngImpMetodo --> D_EsTrans2
    
    D_EsTrans2 -->|Sí| U_IngRefTrans2
    U_IngRefTrans2 --> S_ValRef2
    S_ValRef2 --> S_RegPagoCuota
    
    D_EsTrans2 -->|No| S_RegPagoCuota
    
    S_RegPagoCuota --> S_ShowPagoCuota
    S_ShowPagoCuota --> Fin

    %% ==========================================
    %% 5. MODIFICAR FACTURA
    %% ==========================================
    D_Opcion -->|Modificar factura| U_SelFactMod
    U_SelFactMod --> S_ValPerm5
    S_ValPerm5 -->|No| S_NoAuth
    S_ValPerm5 -->|Sí| S_ConsultFactMod
    S_ConsultFactMod --> S_ValEdit
    
    S_ValEdit -->|No| S_ErrModif
    S_ErrModif --> Fin
    
    S_ValEdit -->|Sí| U_IngModif
    U_IngModif --> S_ValModif
    S_ValModif --> S_UpdFact
    S_UpdFact --> S_ShowFactUpd
    S_ShowFactUpd --> Fin

    %% Conexión de denegación al fin
    S_NoAuth --> Fin
